#!/usr/bin/env bash # kanboard at https://toprakkilic.com/kanboard/ — rerun with a new VERSION to upgrade set -euo pipefail VERSION=1.2.54 # php (kanboard needs it, the rest of the site stays static) apt install -y cron php8.4-fpm php8.4-cli php8.4-gd php8.4-mbstring php8.4-xml php8.4-sqlite3 php8.4-curl php8.4-opcache # code: replaced on every run, root-owned so php can't modify it rm -rf /srv/www/kanboard curl -fsSL "https://github.com/kanboard/kanboard/archive/refs/tags/v$VERSION.tar.gz" | tar xz -C /srv/www mv "/srv/www/kanboard-$VERSION" /srv/www/kanboard chown -R root:root /srv/www/kanboard # data: outside the website folder, kept across upgrades install -d -m 750 -o www-data -g www-data /var/lib/kanboard echo " /srv/www/kanboard/config.php # first run only: set the admin password (replaces admin/admin) if [ ! -f /var/lib/kanboard/db.sqlite ]; then cd /srv/www/kanboard && sudo -u www-data php cli user:reset-password admin fi # daily cronjob (overdue notifications, stats) echo "0 8 * * * www-data cd /srv/www/kanboard && php cli cronjob >/dev/null 2>&1" > /etc/cron.d/kanboard # caddy: Caddyfile with the /kanboard route comes from the repo sudo -u admin git -C /srv/www/toprakkilic.com pull install -m 640 -o root -g caddy /srv/www/toprakkilic.com/config/Caddyfile /etc/caddy/Caddyfile caddy validate --config /etc/caddy/Caddyfile systemctl restart php8.4-fpm systemctl reload caddy echo "done... https://toprakkilic.com/kanboard/"
